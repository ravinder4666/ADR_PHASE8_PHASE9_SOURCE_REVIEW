from __future__ import annotations

import hashlib
import json
import mimetypes
import re
from http.server import BaseHTTPRequestHandler, ThreadingHTTPServer
from pathlib import Path
from urllib.parse import unquote, urlsplit

from engine.orchestrator.phase7_unified_job_service import SERVICE

ROOT = Path(__file__).resolve().parent
UI_ROOT = (ROOT / "ui").resolve()
IMAGE_ROOT = (ROOT / "test_images").resolve()
REPORT_ROOT = (ROOT / "reports" / "phase7").resolve()

# Compatibility alias for existing integrations.
PHASE7_JOBS = SERVICE

CAPABILITIES = {
    "api": True,
    "dashboard_serving": True,
    "read_only_image_acquisition": True,
    "live_scan": True,
    "async_jobs": True,
    "native_structural_validation": True,
    "live_recovery": False,
    "verified_end_to_end_pipeline": False,
    "deleted_file_reconstruction": False,
    "hash_verified_recovery": False,
}

STATIC_FILES = {
    "/": "recovery_dashboard.html",
    "/recovery_dashboard.html": "recovery_dashboard.html",
    "/recovery_dashboard.css": "recovery_dashboard.css",
    "/recovery_dashboard.js": "recovery_dashboard.js",
}
MAX_REQUEST_BYTES = 4096


class ADRHandler(BaseHTTPRequestHandler):
    server_version = "ADRLocalAPI/1.0"

    def log_message(self, fmt, *args):
        print("[ADR API] " + (fmt % args))

    def _send(self, status, payload, content_type="application/json; charset=utf-8"):
        if isinstance(payload, (dict, list)):
            body = json.dumps(payload, ensure_ascii=False).encode("utf-8")
        elif isinstance(payload, str):
            body = payload.encode("utf-8")
        else:
            body = payload
        self.send_response(status)
        self.send_header("Content-Type", content_type)
        self.send_header("Content-Length", str(len(body)))
        self.send_header("X-Content-Type-Options", "nosniff")
        self.send_header("Cache-Control", "no-store")
        self.end_headers()
        self.wfile.write(body)

    def _error(self, status, message, code=None):
        data = {"ok": False, "error": message}
        if code:
            data["code"] = code
        self._send(status, data)

    def _read_json(self):
        raw_length = self.headers.get("Content-Length", "")
        try:
            length = int(raw_length)
        except (TypeError, ValueError):
            raise ValueError("Valid Content-Length is required")
        if length < 0 or length > MAX_REQUEST_BYTES:
            raise OverflowError("Request body exceeds 4096 bytes")
        raw = self.rfile.read(length)
        try:
            value = json.loads(raw.decode("utf-8"))
        except (UnicodeDecodeError, json.JSONDecodeError):
            raise ValueError("Request body must be valid JSON")
        if not isinstance(value, dict):
            raise ValueError("JSON object required")
        return value

    def _job_id(self, path, suffix=""):
        pattern = r"^/api/jobs/([a-fA-F0-9]{32})" + re.escape(suffix) + r"$"
        match = re.fullmatch(pattern, path)
        return match.group(1) if match else None

    def _serve_static(self, path):
        name = STATIC_FILES.get(path)
        if not name:
            return False
        target = (UI_ROOT / name).resolve()
        try:
            target.relative_to(UI_ROOT)
        except ValueError:
            self._error(403, "Invalid static resource path")
            return True
        if not target.is_file():
            self._error(404, "Dashboard asset not found: " + name)
            return True
        content_type = mimetypes.guess_type(target.name)[0] or "application/octet-stream"
        if content_type.startswith("text/") or content_type in ("application/javascript",):
            content_type += "; charset=utf-8"
        self._send(200, target.read_bytes(), content_type)
        return True

    def do_GET(self):
        path = unquote(urlsplit(self.path).path)

        if path == "/api/health":
            self._send(200, {
                "ok": True, "mode": "local_api_foundation",
                "service": "AdvancedDataRecovery",
                "status": "healthy",
            })
            return

        if path == "/api/capabilities":
            self._send(200, {"ok": True, "capabilities": CAPABILITIES})
            return

        if path == "/api/images":
            images = []
            IMAGE_ROOT.mkdir(parents=True, exist_ok=True)
            for item in sorted(IMAGE_ROOT.iterdir(), key=lambda p: p.name.lower()):
                try:
                    if item.is_symlink() or not item.is_file():
                        continue
                    size = item.stat().st_size
                    if size <= 0:
                        continue
                    images.append({
                        "name": item.name,
                        "size_bytes": size,
                        "size": size,
                    })
                except OSError:
                    continue
            self._send(200, {"ok": True, "images": images, "count": len(images)})
            return

        job_id = self._job_id(path)
        if job_id:
            try:
                self._send(200, {"ok": True, "job": PHASE7_JOBS.get(job_id)})
            except KeyError:
                self._error(404, "Job not found", "job_not_found")
            return

        report_job_id = self._job_id(path, "/report")
        if report_job_id:
            try:
                job = PHASE7_JOBS.get(report_job_id)
            except KeyError:
                self._error(404, "Job not found", "job_not_found")
                return
            if job.get("status") != "COMPLETED":
                self._error(409, "Report is available only after a completed scan",
                            "report_not_ready")
                return
            report_path = (REPORT_ROOT / (report_job_id + ".json")).resolve()
            try:
                report_path.relative_to(REPORT_ROOT)
            except ValueError:
                self._error(403, "Invalid report path")
                return
            if not report_path.is_file():
                self._error(404, "Completed job report not found", "report_not_found")
                return
            try:
                report = json.loads(report_path.read_text(encoding="utf-8"))
            except (OSError, json.JSONDecodeError):
                self._error(500, "Report could not be read", "report_read_error")
                return
            self._send(200, {"ok": True, "report": report})
            return

        if path == "/api/jobs":
            self._send(200, {"ok": True, "jobs": PHASE7_JOBS.list_jobs()})
            return

        if self._serve_static(path):
            return
        self._error(404, "Endpoint not found", "not_found")

    def do_POST(self):
        path = unquote(urlsplit(self.path).path)
        try:
            payload = self._read_json()
        except OverflowError as exc:
            self._error(413, str(exc), "request_too_large")
            return
        except ValueError as exc:
            self._error(400, str(exc), "invalid_request")
            return

        if path == "/api/recoveries":
            try:
                from engine.orchestrator.phase9_recovery_api import recover_candidate
                result = recover_candidate(payload)
                self._send(200, {"ok": True, "recovery": result})
            except FileNotFoundError:
                self._error(404, "Source image not found", "image_not_found")
            except FileExistsError:
                self._error(409, "Destination already exists; overwrite refused", "destination_exists")
            except (ValueError, TypeError, KeyError) as exc:
                self._error(400, str(exc), "invalid_recovery_request")
            except Exception:
                self._error(500, "Recovery failed; inspect server logs", "recovery_failed")
            return

        if path == "/api/scans":
            try:
                job = PHASE7_JOBS.create(payload)
                self._send(202, {"ok": True, "job": job})
            except ValueError as exc:
                self._error(400, str(exc), "invalid_scan_request")
            except FileNotFoundError:
                self._error(400, "Image is unavailable", "image_unavailable")
            except Exception as exc:
                self._error(500, "Unable to create scan job: " + type(exc).__name__,
                            "job_creation_failed")
            return

        cancel_job_id = self._job_id(path, "/cancel")
        if cancel_job_id:
            try:
                self._send(200, {
                    "ok": True,
                    "job": PHASE7_JOBS.cancel(cancel_job_id),
                })
            except KeyError:
                self._error(404, "Job not found", "job_not_found")
            return

        if path == "/api/image/inspect":
            name = payload.get("image")
            if not isinstance(name, str) or not name.strip() or name in {".", ".."} \
                    or "/" in name or "\\" in name or "\x00" in name:
                self._error(400, "image must be a filename inside test_images",
                            "invalid_image")
                return
            candidate = (IMAGE_ROOT / name).resolve()
            try:
                candidate.relative_to(IMAGE_ROOT)
            except ValueError:
                self._error(400, "Image path is outside test_images", "invalid_image")
                return
            if candidate.is_symlink() or not candidate.is_file():
                self._error(400, "Image not found", "image_unavailable")
                return
            try:
                digest = hashlib.sha256()
                with candidate.open("rb") as stream:
                    for chunk in iter(lambda: stream.read(1024 * 1024), b""):
                        digest.update(chunk)
                size_bytes = candidate.stat().st_size
                sha256 = digest.hexdigest()
                self._send(200, {
                    "ok": True,
                    "inspection": {
                        "name": candidate.name,
                        "size": size_bytes,
                        "operation": "metadata_only_read_only",
                        "sha256": sha256,
                        "source_modified": False,
                        "inspection_only": True,
                        "recoverability_proven": False,
                    },
                    "image": {
                        "name": candidate.name,
                        "size_bytes": size_bytes,
                        "sha256": sha256,
                        "source_modified": False,
                        "inspection_only": True,
                        "recoverability_proven": False,
                    },
                })
            except OSError:
                self._error(500, "Image inspection failed", "inspection_failed")
            return

        self._error(404, "Endpoint not found", "not_found")


def main():
    IMAGE_ROOT.mkdir(parents=True, exist_ok=True)
    REPORT_ROOT.mkdir(parents=True, exist_ok=True)
    server = ThreadingHTTPServer(("127.0.0.1", 8765), ADRHandler)
    print("Advanced Data Recovery API: http://127.0.0.1:8765")
    print("Dashboard: http://127.0.0.1:8765/")
    print("Read-only scan enabled; verified file recovery remains disabled.")
    try:
        server.serve_forever()
    except KeyboardInterrupt:
        pass
    finally:
        server.server_close()


if __name__ == "__main__":
    main()

"""Create-only atomic output writer; refuses to replace existing destinations."""
import os
import tempfile
from pathlib import Path

def write_recovery_output(destination, data: bytes) -> Path:
    target = Path(destination).resolve()
    target.parent.mkdir(parents=True, exist_ok=True)
    if target.exists():
        raise FileExistsError(str(target))
    fd, temp_name = tempfile.mkstemp(prefix=".recovery-", dir=str(target.parent))
    try:
        with os.fdopen(fd, "wb") as stream:
            stream.write(data)
            stream.flush()
            os.fsync(stream.fileno())
        # Hard-link creation is atomic and fails if target already exists.
        os.link(temp_name, target)
        return target
    finally:
        try:
            os.unlink(temp_name)
        except FileNotFoundError:
            pass

"""Verify an already-created output file; never creates or alters that file."""
import hashlib
from pathlib import Path


def verify_recovered_output(path, expected_size=None, expected_sha256=None,
                            chunk_size=1024 * 1024):
    if not isinstance(chunk_size, int) or isinstance(chunk_size, bool) or chunk_size <= 0:
        raise ValueError("chunk_size must be a positive integer")

    p = Path(path)
    errors = []
    actual_size = None
    actual_hash = None

    try:
        if not p.is_file():
            errors.append("output is not a regular file")
        else:
            actual_size = p.stat().st_size
            digest = hashlib.sha256()
            with p.open("rb") as stream:
                while True:
                    block = stream.read(chunk_size)
                    if not block:
                        break
                    digest.update(block)
            actual_hash = digest.hexdigest()
    except OSError as exc:
        errors.append(f"output verification failed: {type(exc).__name__}")

    if actual_size is not None and expected_size is not None:
        if actual_size != expected_size:
            errors.append("size mismatch")

    if expected_sha256 is not None:
        if not isinstance(expected_sha256, str) or len(expected_sha256) != 64:
            errors.append("expected_sha256 must be a 64-character hex digest")
        elif any(c not in "0123456789abcdefABCDEF" for c in expected_sha256):
            errors.append("expected_sha256 is not hexadecimal")
        elif actual_hash is not None and actual_hash.lower() != expected_sha256.lower():
            errors.append("SHA-256 mismatch")

    return {
        "verified": not errors and actual_size is not None and actual_hash is not None,
        "path": str(p),
        "actual_size": actual_size,
        "sha256": actual_hash,
        "errors": errors,
    }

"""Conservative deleted-NTFS-file discovery and independently verified recovery."""
from __future__ import annotations

import hashlib
from pathlib import Path
from typing import Any

from engine.orchestrator.ntfs_recovery_pipeline import NTFSRecoveryPipeline
from engine.orchestrator.native_ntfs_recovery_service import recover_from_runlist
from engine.filesystem.ntfs_native_bridge import parse_runlist


def _fixup_record(record: bytes, sector_size: int) -> bytes:
    if len(record) < 48 or record[:4] != b"FILE":
        raise ValueError("Invalid NTFS FILE record signature")
    if sector_size <= 0 or len(record) % sector_size:
        raise ValueError("Invalid FILE record or sector geometry")

    usa_offset = int.from_bytes(record[4:6], "little")
    usa_count = int.from_bytes(record[6:8], "little")
    expected = len(record) // sector_size + 1
    if usa_count != expected or usa_offset < 8:
        raise ValueError("Invalid update-sequence array")
    if usa_offset + usa_count * 2 > len(record):
        raise ValueError("Update-sequence array exceeds record boundary")

    result = bytearray(record)
    sequence = record[usa_offset:usa_offset + 2]
    for i in range(1, usa_count):
        trailer = i * sector_size - 2
        replacement = usa_offset + i * 2
        if trailer + 2 > len(record) or replacement + 2 > len(record):
            raise ValueError("Update-sequence fixup boundary violation")
        if record[trailer:trailer + 2] != sequence:
            raise ValueError("NTFS update-sequence fixup mismatch")
        result[trailer:trailer + 2] = record[replacement:replacement + 2]
    return bytes(result)


def parse_deleted_record(record: bytes, sector_size: int = 512) -> dict[str, Any] | None:
    """Parse a deleted regular-file FILE record; None means not eligible."""
    record = _fixup_record(record, sector_size)
    flags = int.from_bytes(record[22:24], "little")
    if flags & 0x01 or flags & 0x02:  # in-use or directory
        return None

    first = int.from_bytes(record[20:22], "little")
    used = int.from_bytes(record[24:28], "little")
    allocated = int.from_bytes(record[28:32], "little")
    if not (48 <= first <= used <= allocated <= len(record)):
        raise ValueError("Invalid FILE record attribute bounds")

    name = None
    data_attr = None
    pos = first
    while pos + 8 <= used:
        attr_type = int.from_bytes(record[pos:pos + 4], "little")
        if attr_type == 0xFFFFFFFF:
            break
        length = int.from_bytes(record[pos + 4:pos + 8], "little")
        if length < 8 or pos + length > used:
            raise ValueError("Invalid NTFS attribute boundary")
        nonresident = record[pos + 8]
        name_length = record[pos + 9]
        name_offset = int.from_bytes(record[pos + 10:pos + 12], "little")
        attr_name = ""
        if name_length:
            begin = pos + name_offset
            end = begin + name_length * 2
            if name_offset < 16 or end > pos + length:
                raise ValueError("Invalid NTFS attribute-name boundary")
            attr_name = record[begin:end].decode("utf-16le", errors="strict")

        if attr_type == 0x30 and nonresident == 0:
            value_length = int.from_bytes(record[pos + 16:pos + 20], "little")
            value_offset = int.from_bytes(record[pos + 20:pos + 22], "little")
            begin = pos + value_offset
            end = begin + value_length
            if value_offset < 24 or end > pos + length:
                raise ValueError("Invalid FILE_NAME attribute boundary")
            value = record[begin:end]
            if len(value) >= 66:
                chars = value[64]
                if 66 + chars * 2 <= len(value):
                    candidate = value[66:66 + chars * 2].decode("utf-16le")
                    if candidate and not attr_name:
                        name = candidate

        elif attr_type == 0x80 and nonresident == 1 and not attr_name:
            if data_attr is not None:
                raise ValueError("Multiple unnamed non-resident DATA attributes")
            if length < 64:
                raise ValueError("Truncated non-resident DATA attribute")
            run_offset = int.from_bytes(record[pos + 32:pos + 34], "little")
            real_size = int.from_bytes(record[pos + 48:pos + 56], "little")
            if run_offset < 64 or run_offset >= length:
                raise ValueError("Invalid DATA runlist offset")
            raw = record[pos + run_offset:pos + length]
            # Only a zero header terminates a runlist; fields may contain zero bytes.
            cursor = 0
            terminated = False
            while cursor < len(raw):
                header = raw[cursor]
                if header == 0:
                    raw = raw[:cursor + 1]
                    terminated = True
                    break
                length_width = header & 0x0F
                offset_width = header >> 4
                if length_width == 0 or length_width > 8 or offset_width > 8:
                    raise ValueError('Invalid runlist field widths')
                entry_end = cursor + 1 + length_width + offset_width
                if entry_end > len(raw):
                    raise ValueError('Truncated runlist entry')
                cursor = entry_end
            if not terminated:
                raise ValueError('DATA runlist terminator missing')
            data_attr = {"real_size": real_size, "runlist": raw}

        pos += length

    if not name or data_attr is None or data_attr["real_size"] <= 0:
        return None

    parsed = parse_runlist(data_attr["runlist"], max_runs=4096)
    if not parsed or any(not isinstance(r, dict) for r in parsed):
        raise ValueError("DATA runlist failed validation")
    if any(r.get("sparse") for r in parsed):
        raise ValueError("Sparse deleted-file data is not supported by this path")
    if any(not isinstance(r.get("lcn"), int) or r["lcn"] < 0
           or not isinstance(r.get("clusters"), int) or r["clusters"] <= 0
           for r in parsed):
        raise ValueError("Invalid physical DATA run")

    return {
        "name": name,
        "flags": flags,
        "real_size": data_attr["real_size"],
        "runlist": data_attr["runlist"],
        "run_count": len(parsed),
        "scope": "deleted FILE record with a parseable non-resident unnamed DATA attribute",
    }


def discover_deleted_files(
    image_path: str,
    *,
    partition_offset: int = 0,
    max_records: int = 100_000,
    max_output_size: int = 64 * 1024 * 1024,
) -> dict[str, Any]:
    """Find candidate deleted regular files in the bounded discovered MFT region."""
    pipeline = NTFSRecoveryPipeline(
        image_path, partition_offset=partition_offset,
        max_output_size=max_output_size
    )
    inspection = pipeline.inspect_volume()
    if not inspection.get("passed"):
        raise ValueError("NTFS volume inspection failed: " + str(inspection.get("error")))

    discovery = pipeline.discover_mft_records(max_records=max_records)
    if not discovery.get("passed"):
        raise ValueError("MFT discovery failed: " + str(discovery.get("error")))

    files = []
    rejected = 0
    for candidate in discovery["candidates"]:
        record = pipeline.image.read_at(candidate["offset"], candidate["record_size"])
        if len(record) != candidate["record_size"]:
            raise IOError("Short read while examining MFT candidate")
        try:
            item = parse_deleted_record(record, pipeline.geometry.bytes_per_sector)
        except (ValueError, UnicodeError):
            rejected += 1
            continue
        if item is None:
            continue
        item["record_offset"] = candidate["offset"]
        item["record_sequence_number"] = candidate["sequence_number"]
        item["record_size"] = candidate["record_size"]
        item["runlist_hex"] = item.pop("runlist").hex()
        files.append(item)

    return {
        "passed": True,
        "candidate_count": discovery["candidate_count"],
        "deleted_file_count": len(files),
        "rejected_malformed_records": rejected,
        "files": files,
        "source_modified": False,
        "scope": (
            "Bounded scan of the discovered MFT region only. "
            "MFT extent mapping, filename parent-directory validation, "
            "and complete deleted-file ownership are not established."
        ),
    }


def recover_deleted_file(
    image_path: str,
    destination: str,
    *,
    record_offset: int,
    expected_sha256: str,
    partition_offset: int = 0,
    max_records: int = 100_000,
    max_output_size: int = 64 * 1024 * 1024,
) -> dict[str, Any]:
    """Recover one discovered candidate only when a trusted independent hash matches."""
    if not isinstance(record_offset, int) or record_offset < 0:
        raise ValueError("Invalid MFT record offset")
    listing = discover_deleted_files(
        image_path, partition_offset=partition_offset,
        max_records=max_records, max_output_size=max_output_size
    )
    matches = [f for f in listing["files"] if f["record_offset"] == record_offset]
    if len(matches) != 1:
        raise ValueError("Selected offset is not one uniquely eligible deleted-file candidate")
    item = matches[0]

    result = recover_from_runlist(
        image_path, destination, bytes.fromhex(item["runlist_hex"]),
        real_size=item["real_size"], expected_sha256=expected_sha256,
        partition_offset=partition_offset, max_output_size=max_output_size
    )
    return {
        **result,
        "file_name_candidate": item["name"],
        "mft_record_offset": record_offset,
        "run_count": item["run_count"],
        "mft_to_file_ownership_proven": False,
        "scope": (
            "Independent hash verified for bytes extracted from the selected candidate's "
            "runlist. Filename/parent ownership and general deleted-file recoverability "
            "are not proven by this result."
        ),
    }

"""Isolated integration of the NTFS runlist parser and safe recovery pipeline."""
from __future__ import annotations

import hashlib
import os
import tempfile
from pathlib import Path
from types import SimpleNamespace

from engine.filesystem.ntfs_native_bridge import parse_runlist
from engine.orchestrator.ntfs_recovery_pipeline import NTFSRecoveryPipeline


def _sha256_file(path: Path) -> str:
    digest = hashlib.sha256()
    with path.open("rb") as stream:
        for block in iter(lambda: stream.read(1024 * 1024), b""):
            digest.update(block)
    return digest.hexdigest()


def _validated_runs(raw_runlist: bytes, max_runs: int = 4096):
    if not isinstance(raw_runlist, bytes) or not raw_runlist:
        raise ValueError("A non-empty raw NTFS runlist is required")
    parsed = parse_runlist(raw_runlist, max_runs=max_runs)
    if not parsed or len(parsed) > max_runs:
        raise ValueError("Runlist is empty or exceeds configured limit")

    runs = []
    for item in parsed:
        if not isinstance(item, dict):
            raise ValueError("Unexpected runlist parser result")
        lcn, clusters = item.get("lcn"), item.get("clusters")
        if item.get("sparse", False):
            raise ValueError("Sparse extents require a separate sparse-data policy")
        if not isinstance(lcn, int) or isinstance(lcn, bool) or lcn < 0:
            raise ValueError("Invalid physical LCN")
        if not isinstance(clusters, int) or isinstance(clusters, bool) or clusters <= 0:
            raise ValueError("Invalid run cluster count")
        runs.append(SimpleNamespace(lcn=lcn, clusters=clusters, sparse=False))
    return tuple(runs)


def recover_from_runlist(
    image_path: str,
    destination: str,
    raw_runlist: bytes,
    *,
    real_size: int,
    expected_sha256: str,
    partition_offset: int = 0,
    max_output_size: int = 64 * 1024 * 1024,
) -> dict:
    """Recover from caller-supplied raw runlist and independently verify content.

    This function does not discover a deleted file's MFT record or prove that
    the supplied runlist belongs to that file. The expected hash must be
    independently trusted.
    """
    source = Path(image_path).resolve(strict=True)
    target = Path(destination).resolve()

    if source == target:
        raise ValueError("Destination cannot be the source image")
    if target.exists():
        raise FileExistsError(f"Refusing to overwrite: {target}")
    if not isinstance(real_size, int) or isinstance(real_size, bool):
        raise ValueError("real_size must be an integer")
    if real_size <= 0 or real_size > max_output_size:
        raise ValueError("real_size is outside the configured output limit")
    if not isinstance(partition_offset, int) or partition_offset < 0:
        raise ValueError("Invalid partition offset")
    if (not isinstance(expected_sha256, str)
        or len(expected_sha256) != 64
        or any(c not in "0123456789abcdefABCDEF" for c in expected_sha256)):
        raise ValueError("A valid independent SHA-256 is required")

    runs = _validated_runs(raw_runlist)
    target.parent.mkdir(parents=True, exist_ok=True)
    source_hash_before = _sha256_file(source)

    fd, temp_name = tempfile.mkstemp(
        prefix=".adr-native-integration-", suffix=".tmp", dir=str(target.parent)
    )
    os.close(fd)
    temp = Path(temp_name)
    temp.unlink()

    try:
        pipeline = NTFSRecoveryPipeline(
            source,
            partition_offset=partition_offset,
            max_output_size=max_output_size,
        )
        inspection = pipeline.inspect_volume()
        if not inspection.get("passed"):
            raise ValueError(
                "NTFS volume inspection failed: "
                + str(inspection.get("error", inspection))
            )

        # Recheck geometry and physical extent boundaries before extraction.
        geometry = pipeline.geometry
        if geometry is None or geometry.cluster_size <= 0:
            raise ValueError("Validated NTFS cluster geometry is unavailable")
        image_size = source.stat().st_size
        for run in runs:
            start = partition_offset + run.lcn * geometry.cluster_size
            end = start + run.clusters * geometry.cluster_size
            if end < start or end > image_size:
                raise ValueError("Runlist extent exceeds source image boundary")

        result = pipeline.extract_verified_bytes(runs, real_size, temp)
        if not result.get("passed") or not temp.is_file():
            raise ValueError("Recovery pipeline did not verify its output")

        actual_size = temp.stat().st_size
        actual_hash = _sha256_file(temp)
        if actual_size != real_size:
            raise ValueError("Recovered file size does not match expected size")
        if actual_hash.lower() != expected_sha256.lower():
            raise ValueError("Independent expected SHA-256 mismatch")

        if _sha256_file(source) != source_hash_before:
            raise RuntimeError("Source image changed during recovery")

        # Same-directory hard link publishes atomically and won't overwrite.
        os.link(temp, target)
        return {
            "passed": True,
            "status": "INDEPENDENT_HASH_VERIFIED",
            "path": str(target),
            "size": actual_size,
            "sha256": actual_hash,
            "source_unchanged": True,
            "native_runlist_parser_available": True,
            "recoverability_proven": True,
            "scope": (
                "Supplied runlist and independent expected hash only; "
                "MFT-to-runlist ownership is not independently established."
            ),
        }
    finally:
        try:
            temp.unlink(missing_ok=True)
        except OSError:
            pass

    FILE_NAME_ATTRIBUTE = 0x30
    DATA_ATTRIBUTE = 0x80
    END_ATTRIBUTE = 0xFFFFFFFF

    def __init__(self, image_path, geometry, partition_offset=0,
                 max_output_size=64 * 1024 * 1024):
        self.image_path = Path(image_path).resolve()
        if isinstance(partition_offset, bool) or not isinstance(partition_offset, int) or partition_offset < 0:
            raise ValueError("partition_offset must be a non-negative integer")
        if isinstance(max_output_size, bool) or not isinstance(max_output_size, int) or max_output_size <= 0:
            raise ValueError("max_output_size must be positive")
        for field in ("bytes_per_sector", "cluster_size", "mft_record_size"):
            value = getattr(geometry, field, None)
            if isinstance(value, bool) or not isinstance(value, int) or value <= 0:
                raise ValueError(f"invalid volume geometry field: {field}")

        self.geometry = geometry
        self.partition_offset = partition_offset
        self.max_output_size = max_output_size
        self.image = DiskImageWorkflowAdapter(self.image_path)
        self.mft_parser = NTFSMFTParser()
        self.attribute_parser = NTFSAttributeParser()
        self.run_parser = NTFSDataRunParser()
        self.filename_parser = NTFSFileNameAttributeParser()

    def parse_record_at(self, offset: int) -> dict[str, Any]:
        """Parse one MFT record at an absolute image offset."""
        if isinstance(offset, bool) or not isinstance(offset, int) or offset < self.partition_offset:
            raise ValueError("record offset is outside the partition range")

        status = self.image.inspect()
        if not status.get("ready"):
            return {"passed": False, "error": status.get("error", "Image unavailable")}

        record_size = self.geometry.mft_record_size
        image_size = status.get("size")
        if not isinstance(image_size, int) or offset + record_size > image_size:
            return {"passed": False, "error": "MFT record exceeds image bounds"}

        raw = self.image.read_at(offset, record_size)
        try:
            record = self.mft_parser.parse(
                raw, sector_size=self.geometry.bytes_per_sector
            )
            attributes = self.attribute_parser.parse(
                record.fixed_record,
                record.first_attribute_offset,
                record.used_size,
            )
        except Exception as exc:
            return {
                "passed": False,
                "offset": offset,
                "error": f"{type(exc).__name__}: {exc}",
            }

        names = []
        data_attributes = []
        warnings = []

        for attr in attributes:
            if attr.type_id == self.FILE_NAME_ATTRIBUTE and attr.value:
                try:
                    parsed_name = self.filename_parser.parse(attr.value)
                    names.append({
                        "name": parsed_name.name,
                        "parent_record_id": parsed_name.parent_record_id,
                        "parent_sequence_number": parsed_name.parent_sequence_number,
                        "namespace": parsed_name.namespace,
                        "real_size": parsed_name.real_size,
                        "allocated_size": parsed_name.allocated_size,
                        "file_flags": parsed_name.file_flags,
                    })
                except Exception as exc:
                    warnings.append(f"Filename attribute rejected: {exc}")

            elif attr.type_id == self.DATA_ATTRIBUTE:
                if attr.non_resident:
                    try:
                        attr_start = attr.attribute_offset
                        run_offset = attr.runlist_offset
                        if (not isinstance(attr_start, int) or attr_start < 0
                                or not isinstance(run_offset, int) or run_offset < 64):
                            raise ValueError("invalid attribute-relative runlist offset")
                        start = attr_start + run_offset
                        attr_end = attr_start + attr.length
                        if start >= attr_end or attr_end > record.used_size:
                            raise ValueError("runlist exceeds its attribute bounds")
                        run_bytes = record.fixed_record[start:attr_end]
                        runs = self.run_parser.parse(
                            run_bytes, start_vcn=attr.lowest_vcn
                        )
                        data_attributes.append({
                            "resident": False,
                            "real_size": attr.real_size,
                            "lowest_vcn": attr.lowest_vcn,
                            "highest_vcn": attr.highest_vcn,
                            "runs": [
                                {
                                    "vcn": r.vcn,
                                    "lcn": r.lcn,
                                    "clusters": r.clusters,
                                    "sparse": r.sparse,
                                } for r in runs
                            ],
                        })
                    except Exception as exc:
                        warnings.append(f"Data runlist rejected: {exc}")
                else:
                    resident_value = bytes(attr.value or b"")
                    data_attributes.append({
                        "resident": True,
                        "real_size": len(resident_value),
                        "resident_data_available": bool(resident_value),
                        "resident_data_hex": resident_value.hex(),
                        "sha256": hashlib.sha256(resident_value).hexdigest(),
                    })

        in_use = bool(record.flags & 0x0001)
        return {
            "passed": True,
            "offset": offset,
            "sequence_number": record.sequence_number,
            "flags": record.flags,
            "in_use": in_use,
            # A free MFT record alone is not sufficient evidence of a
            # recoverable deleted file; require parsed filename and data metadata.
            "deleted_candidate": (
                not in_use and bool(names) and bool(data_attributes)
            ),
            "filenames": names,
            "data_attributes": data_attributes,
            "warnings": warnings,
            "scope": "MFT metadata parsed; deleted content not yet proven recoverable",
        }

    def discover_and_parse(self, max_records=10000) -> dict[str, Any]:
        """Discover structurally plausible MFT records and parse each one."""
        if isinstance(max_records, bool) or not isinstance(max_records, int) or not 1 <= max_records <= 100000:
            raise ValueError("max_records must be between 1 and 100000")

        from engine.filesystem.mft_discovery_engine import MFTDiscoveryEngine

        status = self.image.inspect()
        if not status.get("ready"):
            return {"passed": False, "error": status.get("error", "Image unavailable")}

        start = (
            self.partition_offset
            + self.geometry.mft_lcn * self.geometry.cluster_size
        )
        end = min(
            status["size"],
            start + max_records * self.geometry.mft_record_size,
        )
        try:
            discovery = MFTDiscoveryEngine(
                self.image,
                sector_size=self.geometry.bytes_per_sector,
                record_size=self.geometry.mft_record_size,
            )
            locations = discovery.discover(
                start=start, end=end, max_records=max_records
            )
        except Exception as exc:
            return {"passed": False, "stage": "discovery",
                    "error": f"{type(exc).__name__}: {exc}"}

        parsed = [self.parse_record_at(item.offset) for item in locations]
        valid = [item for item in parsed if item.get("passed") is True]
        deleted = [item for item in valid if item.get("deleted_candidate") is True]
        return {
            "passed": True,
            "records_discovered": len(locations),
            "records_parsed": len(valid),
            "deleted_metadata_candidates": len(deleted),
            "records": parsed,

        }

    def reconstruct_runs(self, runs, real_size: int, destination,
                         expected_sha256: str | None = None) -> dict[str, Any]:
        """Read cluster extents with strict bounds and create-only output."""
        if isinstance(real_size, bool) or not isinstance(real_size, int) or real_size <= 0:
            raise ValueError("real_size must be a positive integer")
        if real_size > self.max_output_size:
            raise ValueError("real_size exceeds max_output_size")

        if expected_sha256 is not None:
            if (not isinstance(expected_sha256, str)
                    or len(expected_sha256) != 64
                    or any(c not in "0123456789abcdefABCDEF" for c in expected_sha256)):
                raise ValueError("expected_sha256 must be a 64-character hexadecimal digest")

        target = Path(destination).resolve()
        if target == self.image_path:
            raise ValueError("destination must not be the source image")
        if target.exists():
            raise FileExistsError(str(target))
        if not isinstance(runs, (list, tuple)) or not runs:
            raise ValueError("non-empty runlist required")

        status = self.image.inspect()
        if not status.get("ready"):
            raise OSError(status.get("error", "Source image unavailable"))
        image_size = status.get("size")
        cluster_size = self.geometry.cluster_size
        output = bytearray()

        physical_ranges = []
        for run in runs:
            if not isinstance(run, dict):
                raise ValueError("invalid runlist entry")

            clusters = run.get("clusters")
            sparse = run.get("sparse", False)
            lcn = run.get("lcn")

            if (not isinstance(clusters, int) or isinstance(clusters, bool)
                    or clusters <= 0):
                raise ValueError("invalid run cluster count")
            if not isinstance(sparse, bool):
                raise ValueError("invalid sparse-run flag")

            if sparse:
                if lcn is not None:
                    raise ValueError("sparse run must not specify a physical LCN")
            else:
                if (not isinstance(lcn, int) or isinstance(lcn, bool)
                        or lcn < 0):
                    raise ValueError("invalid physical cluster location")
                start_cluster = lcn
                end_cluster = lcn + clusters
                if any(start_cluster < old_end and old_start < end_cluster
                       for old_start, old_end in physical_ranges):
                    raise ValueError("overlapping physical cluster extents")
                physical_ranges.append((start_cluster, end_cluster))
            extent_size = clusters * cluster_size
            remaining = real_size - len(output)
            if remaining <= 0:
                break
            take = min(extent_size, remaining)

            if sparse:
                output.extend(b"\x00" * take)
                continue

            if isinstance(lcn, bool) or not isinstance(lcn, int) or lcn < 0:
                raise ValueError("invalid physical cluster location")
            absolute = self.partition_offset + lcn * cluster_size
            if absolute < self.partition_offset or absolute + take > image_size:
                raise ValueError("extent exceeds source image bounds")
            output.extend(self.image.read_at(absolute, take))

        if len(output) != real_size:
            raise ValueError("runlist did not reconstruct the requested file size")

        data = bytes(output)
        digest = hashlib.sha256(data).hexdigest()
        if expected_sha256 is not None and digest.lower() != expected_sha256.lower():
            raise ValueError("Recovered content does not match expected SHA-256")
        written = write_recovery_output(target, data)
        verification = verify_output(
            written, expected_size=real_size, expected_sha256=digest
        )
        verified = (verification.get("passed") is True or verification.get("valid") is True)
        return {
            "passed": verified,
            "path": str(written),
            "size": len(data),
            "sha256": digest,
            "verification": verification,
            "source_modified": False,
            "scope": "extent reconstruction; file format and deleted-file status require independent validation",
        }

from types import SimpleNamespace
from typing import Any

ROOT = Path(__file__).resolve().parents[2]
IMAGE_ROOT = (ROOT / "test_images").resolve()
OUTPUT_ROOT = ROOT / "recovered_files"


def _sha256(path: Path) -> str:
    digest = hashlib.sha256()
    with path.open("rb") as stream:
        for chunk in iter(lambda: stream.read(1024 * 1024), b""):
            digest.update(chunk)
    return digest.hexdigest()


def recover_candidate(payload: dict[str, Any]) -> dict[str, Any]:
    if not isinstance(payload, dict):
        raise TypeError("Request must be a JSON object")

    image_name = payload.get("image")
    if not isinstance(image_name, str) or not image_name or Path(image_name).name != image_name:
        raise ValueError("'image' must be a filename inside test_images")

    source = (IMAGE_ROOT / image_name).resolve(strict=True)
    if source.parent != IMAGE_ROOT or not source.is_file():
        raise ValueError("Image must be a regular file directly inside test_images")

    record_offset = payload.get("record_offset")
    if isinstance(record_offset, bool) or not isinstance(record_offset, int) or record_offset < 0:
        raise ValueError("'record_offset' must be a non-negative integer")

    geometry_data = payload.get("geometry")
    if not isinstance(geometry_data, dict):
        raise ValueError("'geometry' must contain validated NTFS geometry")
    required = ("bytes_per_sector", "cluster_size", "mft_record_size", "mft_lcn")
    values = {}
    for key in required:
        value = geometry_data.get(key)
        minimum = 0 if key == "mft_lcn" else 1
        if isinstance(value, bool) or not isinstance(value, int) or value < minimum:
            raise ValueError(f"geometry.{key} is invalid")

    if (not isinstance(destination_name, str) or not destination_name
            or Path(destination_name).name != destination_name
            or destination_name in (".", "..")):
        raise ValueError("'destination' must be a plain output filename")

    expected = payload.get("expected_sha256")
    if expected is not None:
        if not isinstance(expected, str) or not re.fullmatch(r"[0-9a-fA-F]{64}", expected):
            raise ValueError("'expected_sha256' must be a 64-character SHA-256 digest")
        expected = expected.lower()


    OUTPUT_ROOT.mkdir(parents=True, exist_ok=True)
    destination = (OUTPUT_ROOT / destination_name).resolve()
    if destination.parent != OUTPUT_ROOT.resolve():
        raise ValueError("Destination must remain inside recovered_files")
    if destination == source or destination.exists():
        raise FileExistsError(str(destination))

    from engine.orchestrator.ntfs_recovery_integration import NTFSRecoveryIntegration

    geometry = SimpleNamespace(**values)
    source_hash_before = _sha256(source)
    engine = NTFSRecoveryIntegration(
        image_path=source,
        geometry=geometry,
        partition_offset=partition_offset,
    )

    result = engine.recover_deleted_record(
        record_offset=record_offset,
        destination=destination,
        expected_sha256=expected,
    )

    source_hash_after = _sha256(source)
    source_unchanged = source_hash_before == source_hash_after
    output_exists = destination.is_file()
    output_hash = _sha256(destination) if output_exists else None
    output_size = destination.stat().st_size if output_exists else None

    passed = (
        bool(result.get("passed", result.get("valid", False)))
        if isinstance(result, dict) else bool(result)
    )
    reported_size = result.get("size") if isinstance(result, dict) else None
    known_hash_match = bool(expected is not None and output_hash == expected and output_exists)

    if not source_unchanged:
        recovery_status = "SOURCE_MODIFIED"
        status_reason = "Source image hash changed during recovery"
    elif not output_exists:
        recovery_status = "OUTPUT_MISSING" if passed else "RECOVERY_FAILED"
        status_reason = "Engine reported success but no output exists" if passed else "No recovery output was produced"
    elif isinstance(reported_size, int) and not isinstance(reported_size, bool) and output_size != reported_size:
        recovery_status = "INCOMPLETE_OUTPUT" if output_size < reported_size else "DAMAGED_OUTPUT"
        status_reason = "Output size differs from the engine-reported recovered size"
    elif expected is None:
        recovery_status = "UNVERIFIABLE_OUTPUT"
        status_reason = "No independently known original SHA-256 was supplied"
    elif not known_hash_match:
        recovery_status = "DAMAGED_OUTPUT"
        status_reason = "Output does not match the independently supplied expected SHA-256"
    elif not passed:
        recovery_status = "RECOVERY_FAILED"
        status_reason = "Engine did not report successful recovery and verification"
    else:
        recovery_status = "VERIFIED_OUTPUT"
        status_reason = "Engine passed, source is unchanged, and output matches the expected SHA-256"

    return {
        "status": recovery_status,
        "status_reason": status_reason,
        "engine_result": result,
        "source_sha256_before": source_hash_before,
        "source_sha256_after": source_hash_after,
        "source_unchanged": source_unchanged,
        "output_path": str(destination) if output_exists else None,
        "output_size": output_size,
        "engine_reported_size": reported_size,
        "output_sha256": output_hash,
        "expected_sha256_supplied": expected is not None,
        "expected_sha256_match": known_hash_match,
        "known_original_hash_verified": known_hash_match,
        "limitations": [
            "Only images located directly inside test_images are accepted",
            "Geometry must be supplied from a trusted NTFS inspection",
            "A matching hash is meaningful only when the expected hash comes from an independent trusted source",
            "A synthetic-image test does not establish real-world recovery capability",
        ],
    }

"""Integrity gate: only complete, bounded, hash-matching bytes can be approved."""
from dataclasses import dataclass
import hashlib

@dataclass(frozen=True)
class IntegrityDecision:
    approved: bool
    size: int
    sha256: str | None
    reasons: tuple[str, ...]

def evaluate_recovery(data: bytes, *, expected_size: int, expected_sha256: str | None,
                      metadata_consistent: bool, extents_complete: bool,
                      source_read_only: bool = True) -> IntegrityDecision:
    reasons = []
    if not isinstance(data, bytes):
        return IntegrityDecision(False, 0, None, ("recovered data must be bytes",))
    digest = hashlib.sha256(data).hexdigest()
    if not isinstance(expected_size, int) or expected_size < 0 or len(data) != expected_size:
        reasons.append("recovered byte count does not match expected size")
    if expected_sha256 is not None:
        if not isinstance(expected_sha256, str) or len(expected_sha256) != 64:
            reasons.append("expected SHA-256 must be 64 hexadecimal characters")
        else:
            try:
                int(expected_sha256, 16)
            except ValueError:
                reasons.append("expected SHA-256 is not hexadecimal")
            else:
                if digest.lower() != expected_sha256.lower():
                    reasons.append("SHA-256 mismatch")
    if not metadata_consistent:
        reasons.append("metadata consistency gate failed")
    if not extents_complete:
        reasons.append("extent completeness gate failed")
    if not source_read_only:
        reasons.append("source was not confirmed read-only")
    return IntegrityDecision(not reasons, len(data), digest, tuple(reasons) if reasons else ("integrity checks passed",))

"""Prevents unsafe recovery decisions; it does not perform file writes."""

from pathlib import Path

def evaluate_recovery_safety(source, destination, source_read_only=True,
                             destination_exists=False, overwrite_approved=False):
    errors = []
    if not str(source).strip() or not str(destination).strip():
        errors.append("source and destination are required")
    else:
        try:
            if Path(source).resolve() == Path(destination).resolve():
                errors.append("source and destination resolve to the same path")
        except (OSError, ValueError, RuntimeError):
            errors.append("could not safely resolve source/destination")

    if not source_read_only:
        errors.append("source must remain read-only")
    if destination_exists and not overwrite_approved:
        errors.append("existing destination overwrite is not approved")

    return {
        "safe_to_proceed": not errors,
        "errors": errors,
        "source_write_permitted": False,
        "overwrite_approved": bool(destination_exists and overwrite_approved)
    }

"""Resolve a relative output path without escaping the selected output root."""
from pathlib import Path

def resolve_recovery_path(output_root, relative_path, allow_overwrite=False):
    if relative_path is None:
        raise ValueError("relative output path is required")
    relative_text = str(relative_path)
    if not relative_text.strip():
        raise ValueError("relative output path is required")
    if "\x00" in relative_text:
        raise ValueError("NUL bytes are not allowed in output paths")

    root = Path(output_root).expanduser().resolve()
    raw = Path(relative_text)
    if raw.is_absolute() or raw.drive:
        raise ValueError("absolute paths are not allowed")
    if any(part == ".." for part in raw.parts):
        raise ValueError("parent traversal is not allowed")
    target = (root / raw).resolve()
    try:
        target.relative_to(root)
    except ValueError as exc:
        raise ValueError("output path escapes output root") from exc
    if target == root:
        raise ValueError("output path must name a file below the root")
    if target.exists() and not allow_overwrite:
        raise FileExistsError(str(target))
    return target

def make_filename_attribute(filename="fixture.bin", real_size=8):
    name_bytes = filename.encode("utf-16le")
    value = bytearray(66 + len(name_bytes))

    # Parent file reference: record 5, sequence number 1.
    struct.pack_into("<Q", value, 0, (1 << 48) | 5)
    struct.pack_into("<Q", value, 40, real_size)
    struct.pack_into("<Q", value, 48, real_size)
    struct.pack_into("<I", value, 56, 0)
    value[64] = len(filename)
    value[65] = 1
    value[66:] = name_bytes

    length = (24 + len(value) + 7) & ~7
    attr = bytearray(length)
    struct.pack_into("<I", attr, 0, 0x30)
    struct.pack_into("<I", attr, 4, length)
    attr[8] = 0
    struct.pack_into("<H", attr, 14, 1)
    struct.pack_into("<I", attr, 16, len(value))
    struct.pack_into("<H", attr, 20, 24)
    attr[24:24 + len(value)] = value
    return bytes(attr)


def make_data_attribute(real_size=8):
    # Two one-cluster extents: LCN 1, then LCN 3.
    # NTFS runlist uses signed LCN deltas: +1, then +2.
    runlist = b"\x11\x01\x01\x11\x01\x02\x00"
    attr = bytearray(72)
    struct.pack_into("<I", attr, 0, 0x80)
    struct.pack_into("<I", attr, 4, len(attr))
    attr[8] = 1
    struct.pack_into("<H", attr, 14, 2)
    struct.pack_into("<Q", attr, 16, 0)
    struct.pack_into("<Q", attr, 24, 1)
    struct.pack_into("<H", attr, 32, 64)
    struct.pack_into("<Q", attr, 40, 8)
    struct.pack_into("<Q", attr, 48, real_size)
    attr[64:64 + len(runlist)] = runlist
    return bytes(attr)


def make_deleted_mft_record():
    record = bytearray(1024)
    record[:4] = b"FILE"

    # Update Sequence Array for a 1024-byte record with 512-byte sectors.
    usa_offset = 48
    struct.pack_into("<H", record, 4, usa_offset)
    struct.pack_into("<H", record, 6, 3)
    record[usa_offset:usa_offset + 2] = b"\xA5\x5A"
    record[usa_offset + 2:usa_offset + 4] = b"\x00\x00"
    record[usa_offset + 4:usa_offset + 6] = b"\x00\x00"

    struct.pack_into("<H", record, 16, 7)  # Sequence number
    struct.pack_into("<H", record, 20, 56)
    struct.pack_into("<H", record, 22, 0)  # Not in use: deleted candidate

    filename_attr = make_filename_attribute()
    data_attr = make_data_attribute()
    pos = 56
    record[pos:pos + len(filename_attr)] = filename_attr
    pos += len(filename_attr)
    record[pos:pos + len(data_attr)] = data_attr
    pos += len(data_attr)
    struct.pack_into("<I", record, pos, 0xFFFFFFFF)
    used_size = pos + 4

    struct.pack_into("<I", record, 24, used_size)
    struct.pack_into("<I", record, 28, len(record))

    # Protect each sector trailer with the USA sequence value.
    record[510:512] = b"\xA5\x5A"
    record[1022:1024] = b"\xA5\x5A"
    return bytes(record)

class TestCancellationArbitration(unittest.TestCase):
    def test_queued_cancel_cannot_be_overwritten_by_worker_start(self):
        service = Phase7JobService()
        job_id = "a" * 32
        service._jobs[job_id] = {
            "job_id": job_id,
            "status": "QUEUED",
            "stage": "queued",
            "progress": 0,
            "updated_utc": "",
            "cancel_requested": False,
        }

        service._run(job_id, ROOT / "missing-image.img", 1000)

        final = service.get(job_id)
        self.assertEqual(final["status"], "CANCELLED")
        self.assertEqual(final["stage"], "cancelled")
        self.assertTrue(final["cancel_requested"])


class TestLiveDashboardAPIWorkflow(unittest.TestCase):
    def test_http_scan_poll_and_report_lifecycle(self):
        image_root = ROOT / "test_images"
        image = image_root / "adr_synthetic_ntfs_volume.img"
        if not image.is_file() or image.stat().st_size <= 0:
            self.skipTest("Synthetic NTFS test image is unavailable")

        server = ThreadingHTTPServer(
            ("127.0.0.1", 0), adr_api_server.ADRHandler
        )
        thread = threading.Thread(target=server.serve_forever, daemon=True)
        thread.start()
        base = f"http://127.0.0.1:{server.server_port}"

        def get_json(path):
            with urlopen(base + path, timeout=5) as response:
                return response.status, json.loads(
                    response.read().decode("utf-8")
                )

        def post_json(path, payload):
            request = Request(
                base + path,
                data=json.dumps(payload).encode("utf-8"),
                headers={"Content-Type": "application/json"},
                method="POST",
            )
            with urlopen(request, timeout=10) as response:
                return response.status, json.loads(
                    response.read().decode("utf-8")
                )

        try:
            status_code, health = get_json("/api/health")
            self.assertEqual(status_code, 200)
            self.assertTrue(health["ok"])

            status_code, inventory = get_json("/api/images")
            self.assertEqual(status_code, 200)
            self.assertIn(
                image.name, [item["name"] for item in inventory["images"]]
            )

            status_code, created = post_json(
                "/api/scans",
                {"image": image.name, "max_records": 1000},
            )
            self.assertEqual(status_code, 202)
            self.assertTrue(created["ok"])
            job_id = created["job"]["job_id"]
            self.assertRegex(job_id, r"^[a-fA-F0-9]{32}$")

            deadline = time.monotonic() + 45
            job = created["job"]
            while time.monotonic() < deadline:
                status_code, envelope = get_json(f"/api/jobs/{job_id}")
                self.assertEqual(status_code, 200)
                self.assertTrue(envelope["ok"])
                job = envelope["job"]
                if job["status"] in {"COMPLETED", "FAILED", "CANCELLED"}:
                    break
                time.sleep(0.2)
            else:
                self.fail("Scan did not reach a terminal state within 45 seconds")

            if job["status"] == "COMPLETED":
                status_code, report_envelope = get_json(
                    f"/api/jobs/{job_id}/report"
                )
                self.assertEqual(status_code, 200)
                report = report_envelope["report"]
                self.assertEqual(report["job_id"], job_id)
                self.assertEqual(
                    report["source_sha256"], job["source_sha256"]
                )
                self.assertFalse(report["result"]["recovery_performed"])
                self.assertFalse(report["result"]["recoverability_proven"])
            else:
                # A failed/cancelled scan must not expose a completed report.
                request = Request(base + f"/api/jobs/{job_id}/report")
                from urllib.error import HTTPError
                with self.assertRaises(HTTPError) as caught:
                    urlopen(request, timeout=5)
                self.assertEqual(caught.exception.code, 409)

                self.assertNotEqual(
                    job["status"], "RUNNING",
                    "Terminal job unexpectedly remained RUNNING"
                )

            # Recovery capability remains honestly disabled.
            _, capabilities = get_json("/api/capabilities")
            self.assertFalse(capabilities["capabilities"]["live_recovery"])
            self.assertFalse(
                capabilities["capabilities"]["verified_end_to_end_pipeline"]
            )
        finally:
            server.shutdown()
            server.server_close()
            thread.join(timeout=5)

    unittest.main()

class TestVerifiedRecoveryPipeline(unittest.TestCase):

        calls = []
        pipeline = VerifiedRecoveryPipeline([
            ("scan", lambda ctx: calls.append("scan") or {"passed": True, "count": 2}),
            ("validate", lambda ctx: calls.append("validate") or {"passed": True}),
            ("verify", lambda ctx: calls.append("verify") or {"passed": True}),
        ])
        result = pipeline.run()
        self.assertTrue(result["passed"])
        self.assertEqual(calls, ["scan", "validate", "verify"])

    def test_false_result_stops_later_stages(self):
        calls = []
        pipeline = VerifiedRecoveryPipeline([
            ("scan", lambda ctx: {"passed": True}),
            ("validate", lambda ctx: {"passed": False, "reason": "bad_record"}),
            ("publish", lambda ctx: calls.append("publish") or {"passed": True}),
        ])
        result = pipeline.run()
        self.assertFalse(result["passed"])
        self.assertEqual(calls, [])
        self.assertTrue(result["stages"][2]["details"]["skipped"])

    def test_exception_stops_later_stages(self):
        def broken(_):
            raise OSError("fixture failure")

        pipeline = VerifiedRecoveryPipeline([
            ("read", broken),
            ("publish", lambda ctx: {"passed": True}),
        ])
        result = pipeline.run()
        self.assertFalse(result["passed"])
        self.assertEqual(result["stages"][0]["details"]["error_type"], "OSError")
        self.assertTrue(result["stages"][1]["details"]["skipped"])

        pipeline = VerifiedRecoveryPipeline([

    const cells = [
      label,
      "NTFS/MFT",
      c.record_size == null ? "Unknown" : `${c.record_size} bytes`,
      valid ? "Structural check passed" : (invalid ? "Invalid header" : "Unverifiable"),
      "Not recovered",
      ""
    ];
    for (const value of cells) {
      const td = document.createElement("td");
      td.textContent = String(value);
      tr.appendChild(td);
    }
    body.appendChild(tr);
  }
  if (!filtered.length) {
    const tr = document.createElement("tr");
    const td = document.createElement("td");
    td.colSpan = 6;
    td.textContent = candidates.length ? "No matching candidates." : "No scan candidates yet.";
    tr.appendChild(td);
    body.appendChild(tr);
  }
  if ($("candidateCount")) $("candidateCount").textContent = String(candidates.length);
  if ($("validatedCount"))
    $("validatedCount").textContent = String(candidates.filter(c => c.header_valid === true).length);
  if ($("resultCount")) $("resultCount").textContent = `${filtered.length} candidate(s)`;
}

function applyJob(job) {
  activeJob = job;
  if ($("evidenceSourceName") && job.source_name)
    $("evidenceSourceName").textContent = job.source_name;
  if ($("sourceHashValue")) {
    $("sourceHashValue").textContent = job.source_sha256 || "Hash pending";
    $("sourceHashValue").title = job.source_sha256 || "";
  }
  progress(job);
  status(`${job.status} Ã‚Â· ${job.stage || "scan"} Ã‚Â· ${job.progress || 0}%`);
  const result = job.result || {};
  candidates = Array.isArray(result.candidates) ? result.candidates : [];
  renderCandidates();
  const finished = terminal(job.status);
  if ($("cancelScanBtn")) $("cancelScanBtn").disabled = finished;
  updateScanAvailability();
  // The API serves reports only for COMPLETED jobs; FAILED/CANCELLED are not reportable.
  if ($("reportBtn")) $("reportBtn").disabled = job.status !== "COMPLETED";
  if (finished) {
    clearTimeout(pollTimer);
    if (job.status === "COMPLETED") toast("Scan finished. No files were recovered.");
    else if (job.status === "FAILED") {
      const detail = typeof job.error === "string" ? job.error
        : (job.error && (job.error.message || job.error.type)) || "Unknown scan error";
      toast("Scan failed: " + detail);
    }
    else toast("Scan cancelled.");
  }
}
async function pollJob(id) {
  try {
    const data = await requestJSON(`/api/jobs/${encodeURIComponent(id)}`);
    applyJob(data.job);
    if (!terminal(data.job.status)) pollTimer = setTimeout(() => pollJob(id), 900);
  } catch (e) {
    status("Job status temporarily unavailable: " + e.message + ". Scan remains locked until status is confirmed.");
    if ($("scanBtn")) $("scanBtn").disabled = true;
    if ($("cancelScanBtn")) $("cancelScanBtn").disabled = false;
    pollTimer = setTimeout(() => pollJob(id), 2000);
  }
}

    const jobs = Array.isArray(data.jobs) ? data.jobs : [];
    if (jobs.length > 0) {

  } catch (error) {

  }
}

async function startScan() {
  const image = $("imageSelect")?.value;
  if (!image) { toast("Please select an image first."); return; }
  if (activeJob && !terminal(activeJob.status)) return;
  if (pollTimer) clearTimeout(pollTimer);
  $("scanBtn").disabled = true;
  $("reportBtn").disabled = true;
  $("cancelScanBtn").disabled = false;
  candidates = []; renderCandidates();
  status("Submitting scan...");
  try {
    const data = await requestJSON("/api/scans", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ image, max_records: Number($("maxRecords").value) })
    });
    applyJob(data.job);
    if (!terminal(data.job.status)) pollJob(data.job.job_id);
  } catch (e) {
    if ($("cancelScanBtn")) $("cancelScanBtn").disabled = true;
    activeJob = null;
    updateScanAvailability();
    status("Scan request rejected: " + e.message);
    toast("Scan could not start: " + e.message);
  }
}
async function cancelScan() {
  if (!activeJob?.job_id) return;
  $("cancelScanBtn").disabled = true;
  try {
    const data = await requestJSON(
      `/api/jobs/${encodeURIComponent(activeJob.job_id)}/cancel`,
      { method: "POST", headers: { "Content-Type": "application/json" }, body: "{}" }
    );
    applyJob(data.job);
    if (!terminal(data.job.status)) pollJob(data.job.job_id);
  } catch (e) { toast("Cancellation failed: " + e.message); }
}
async function downloadReport() {
  if (!activeJob?.job_id) {
    toast("Start a scan before downloading its report.");
    return;
  }
  if (activeJob.status !== "COMPLETED") {
    toast("Report is available only after a completed scan.");
    return;
  }
  const button = $("reportBtn");
  if (button) button.disabled = true;
  try {
    const report = await requestJSON(
      `/api/jobs/${encodeURIComponent(activeJob.job_id)}/report`
    );
    const blob = new Blob([JSON.stringify(report, null, 2)], {
      type: "application/json"
    });
    const url = URL.createObjectURL(blob);
    const a = document.createElement("a");
    a.href = url;
    a.download = `adr_scan_${activeJob.job_id}.json`;
    document.body.appendChild(a);
    a.click();
    a.remove();
    setTimeout(() => URL.revokeObjectURL(url), 1000);
    toast("API scan report downloaded.");
  } catch (e) {
    toast("Report download failed: " + e.message);
  } finally {
    if (button) button.disabled = !activeJob || activeJob.status !== "COMPLETED";
  }
}

function adrEscape(value) {
  return String(value ?? "").replace(/[&<>"']/g, ch => ({
    "&":"&amp;", "<":"&lt;", ">":"&gt;", '"':"&quot;", "'":"&#39;"
  })[ch]);
}
function adrStorageRead(key) {
  try {

}

document.addEventListener("submit", async event => {
  const form = event.target;
  if (!(form instanceof HTMLFormElement)) return;
  if (form.id === "adrRecoverForm") {
    event.preventDefault();
    const fd = new FormData(form);
    const target = $("adrRecoveryResult");
    const integerField = (name) => {
      const raw = String(fd.get(name) ?? "").trim();
      if (!/^\d+$/.test(raw)) throw new Error("Invalid integer: " + name);
      const value = Number(raw);
      if (!Number.isSafeInteger(value)) throw new Error("Value is out of range: " + name);
      return value;
    };
    let payload;
    try {
      payload = {
        image: String(fd.get("image") || "").trim(),
        record_offset: integerField("record_offset"),
        geometry: {
          bytes_per_sector: integerField("bytes_per_sector"),
          cluster_size: integerField("cluster_size"),
          mft_record_size: integerField("mft_record_size"),
          mft_lcn: integerField("mft_lcn")
        },
        destination: String(fd.get("destination") || "").trim()
      };
      const partitionOffset = String(fd.get("partition_offset") || "").trim();
      if (partitionOffset !== "") {
        if (!/^\d+$/.test(partitionOffset) || !Number.isSafeInteger(Number(partitionOffset)))
          throw new Error("Invalid partition offset.");
        payload.geometry.partition_offset = Number(partitionOffset);
      }
      const expected = String(fd.get("expected_sha256") || "").trim();
      if (expected) {
        if (!/^[a-fA-F0-9]{64}$/.test(expected))
          throw new Error("Expected SHA-256 must contain exactly 64 hexadecimal characters.");
        payload.expected_sha256 = expected;
      }
      if (!payload.image || !payload.destination)
        throw new Error("Select an image and enter a destination filename.");
      if (/[\\\\/:*?"<>|]/.test(payload.destination) ||
          payload.destination === "." || payload.destination === "..")
        throw new Error("Destination must be a plain filename, not a path.");
    } catch (e) {
      if (target) target.textContent = "Recovery not started: " + e.message;
      return;
    }

    const submit = form.querySelector('button[type="submit"]');
    if (submit) submit.disabled = true;
    if (target) target.textContent = "Recovery request running; checking output and source integrity...";
    try {
      const envelope = await requestJSON("/api/recoveries", {
        method: "POST",
        headers: {"Content-Type": "application/json"},
        body: JSON.stringify(payload)
      });
      const result = envelope.recovery || {};
      const lines = [
        "API request: accepted",
        "Recovery status: " + String(result.status || "UNKNOWN"),
        "Source unchanged: " + String(result.source_unchanged),
        "Known-original hash verified: " + String(result.known_original_hash_verified),
        "Output path: " + String(result.output_path || "not provided"),
        "Output size: " + String(result.output_size ?? "unknown"),
        "Output SHA-256: " + String(result.output_sha256 || "not provided"),
        "Source SHA-256 before: " + String(result.source_sha256_before || "not provided"),
        "Source SHA-256 after: " + String(result.source_sha256_after || "not provided")
      ];
      if (Array.isArray(result.limitations) && result.limitations.length) {
        lines.push("Limitations:");
        for (const item of result.limitations) lines.push("- " + String(item));
      }
      if (target) target.textContent = lines.join("\n");
      toast(result.status === "VERIFIED_OUTPUT"
        ? (result.known_original_hash_verified
            ? "Output verified against the supplied known-original hash."
            : "Output checks passed; original-file identity is not proven.")
        : "Recovery output is NOT VERIFIED.");
    } catch (e) {
      if (target) target.textContent = "Recovery request failed: " + e.message;
      toast("Recovery failed: " + e.message);
    } finally {
      if (submit) submit.disabled = false;
    }
  } else if (form.id === "adrCaseForm") {
    event.preventDefault();
    const fd = new FormData(form);
    const title = String(fd.get("title") || "").trim();
    const notes = String(fd.get("notes") || "").trim();
    if (!title) { toast("Case title is required."); return; }
    const cases = adrStorageRead("adr.dashboard.cases.v1");
    cases.unshift({
      id: "ADR-" + new Date().toISOString().replace(/\D/g, "").slice(0,14),
      title, notes, created: new Date().toLocaleString()
    });
    if (adrStorageWrite("adr.dashboard.cases.v1", cases)) {
      await adrShowView("Create Case"); toast("Case saved in this browser.");
    }
  } else if (form.id === "adrEvidenceForm") {
    event.preventDefault();
    const image = String(new FormData(form).get("image") || "");
    if (!image) { toast("Select an image first."); return; }
    try {
      const item = await adrInspectImage(image, "adrEvidenceResult");
      const entries = adrStorageRead("adr.dashboard.evidence.v1");
      entries.unshift({name:item.name || image, size:item.size, sha256:item.sha256, checked:new Date().toLocaleString()});
      if (adrStorageWrite("adr.dashboard.evidence.v1", entries)) {
        await adrShowView("Evidence & Hashes");
        const target = $("adrEvidenceResult");
        if (target) target.textContent = "Inspection recorded in browser storage. Hash: " + item.sha256;
      }
    } catch (_) {}
  } else if (form.id === "adrImagingForm") {
    event.preventDefault();
    const image = String(new FormData(form).get("image") || "");
    if (image) try { await adrInspectImage(image, "adrImagingResult"); } catch (_) {}
  }
});

document.addEventListener("DOMContentLoaded", async () => {
  new Set([$("scanBtn"), $("startScan"), $("startScanBtn")].filter(Boolean))
    .forEach(button => button.addEventListener("click", startScan));
  $("reportBtn")?.addEventListener("click", downloadReport);
  $("cancelScanBtn")?.addEventListener("click", cancelScan);
  $("imageSelect")?.addEventListener("change", updateScanAvailability);
  $("searchInput")?.addEventListener("input", renderCandidates);
  $("typeFilter")?.addEventListener("change", renderCandidates);
  if ($("reportBtn")) $("reportBtn").disabled = true;
  if ($("cancelScanBtn")) $("cancelScanBtn").disabled = true;
  updateScanAvailability();
  renderCandidates();
  await loadImages();
  await restoreLatestJob();
  try {
    const [health, caps] = await Promise.all([
      requestJSON("/api/health"), requestJSON("/api/capabilities")
    ]);
    if ($("adrApiHealth")) $("adrApiHealth").textContent = health.ok ? "Connected" : "Unhealthy";
    if ($("adrApiCapabilities")) {
      const c = caps.capabilities || {};
      $("adrApiCapabilities").textContent =
        `Read-only scan: ${c.live_scan ? "enabled" : "disabled"} | Recovery: ${c.live_recovery ? "enabled" : "disabled"}`;
    }
  } catch (_) {
    if ($("adrApiHealth")) $("adrApiHealth").textContent = "Disconnected";
  }
});
